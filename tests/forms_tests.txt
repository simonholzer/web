    <script>
        // --- JavaScript Logic ---

        const draggables = document.querySelectorAll('.draggable');
        let activeItem = null;
        let initialX, initialY;
        let currentX, currentY;
        let xOffset = 0, yOffset = 0;

        draggables.forEach(item => {
            // Mouse Events
            item.addEventListener('mousedown', dragStart);
            
            // Touch Events (Mobile)
            item.addEventListener('touchstart', dragStart);
        });

        // Add global listeners to body so dragging doesn't break if you move mouse fast
        document.body.addEventListener('mouseup', dragEnd);
        document.body.addEventListener('touchend', dragEnd);
        document.body.addEventListener('mousemove', drag);
        document.body.addEventListener('touchmove', drag);

        function dragStart(e) {
            if (e.type === "touchstart") {
                initialX = e.touches[0].clientX - xOffset;
                initialY = e.touches[0].clientY - yOffset;
            } else {
                initialX = e.clientX - xOffset;
                initialY = e.clientY - yOffset;
            }

            activeItem = e.currentTarget;
            
            // Get current transform values if they exist
            // This logic is simplified; for a robust app, we would parse the matrix
            // But here we reset offsets based on the element's current computed style position
            // To make "pickup and drop" work seamlessly, we calculate the offset relative to the element rect
            
            const rect = activeItem.getBoundingClientRect();
            
            if (e.type === "touchstart") {
               xOffset = e.touches[0].clientX - rect.left;
               yOffset = e.touches[0].clientY - rect.top;
            } else {
               xOffset = e.clientX - rect.left;
               yOffset = e.clientY - rect.top;
            }
            
            activeItem.classList.add('active');
        }

        function dragEnd(e) {
            if (activeItem) {
                activeItem.classList.remove('active');
                activeItem = null;
            }
        }

        function drag(e) {
            if (activeItem) {
                e.preventDefault();
                
                let clientX, clientY;

                if (e.type === "touchmove") {
                    clientX = e.touches[0].clientX;
                    clientY = e.touches[0].clientY;
                } else {
                    clientX = e.clientX;
                    clientY = e.clientY;
                }

                // Calculate new position
                const newX = clientX - xOffset;
                const newY = clientY - yOffset;

                // Apply new position
                activeItem.style.left = `${newX}px`;
                activeItem.style.top = `${newY}px`;
            }
        }
    </script>
